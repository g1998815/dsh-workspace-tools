**Goal:** 把"变更"页签升级为上下双分区：上半=未提交变更（checkbox 多选 + 单个/全部提交），下半=已提交历史（查看 + 安全回退），两区之间分隔条可拖拽调高；整个右侧工具侧边栏宽度可向左/右拖拽调整。

**Tech Stack:** Node.js ≥ 22（ESM）、`node:test`、esbuild、React 18 + `react/jsx-runtime`、git CLI（`-c core.quotepath=false`）。

**设计决策（本计划定稿）：**
- **写操作边界**（首次）：`git.commit` / `git.reset` 全部走 `guardCwd`（fail-closed sessionId + cwd 匹配）；`git.reset` **只支持 mixed**（用户裁定：安全回退，改动保留工作区），target 必须匹配 `/^[0-9a-f]{4,40}$/`（防注入）；`git.commit` 的 message 经 `execFile` 数组参数传递（无 shell 注入面）。
- **单文件/全部提交**：`git.commit` payload `files?: string[]`——提供则 `git add -- <files…>`（`--` 防选项注入，逐文件），省略则 `git add -A`；无任何可提交变更 → 结构化错误 `nothing-to-commit`（信封内 message 可读，code 走 `internal` 映射）。
- **提交历史**：`git.log` 用 `git log -n <limit> --format=%H%x1f%h%x1f%s%x1f%an%x1f%aI`（unit separator 分隔），host 解析为 `[{hash, shortHash, subject, author, date}]`，client 纯函数做相对时间格式化（"5 分钟前"）。
- **分区拖拽**：上半区高度用百分比 state（默认 50%），分隔条 `data-wt-split` onMouseDown → window mousemove 按容器高度换算百分比（clamp 20–80%）→ mouseup 清理。
- **侧边栏宽度拖拽**：rail 宽度 state（默认 300），rail 左缘 4px 把手 `data-wt-resize`（cursor: col-resize）→ window mousemove 计算 `innerWidth - clientX`（clamp 200–600）。
- **回退交互**：选中某条历史 → 底部按钮区出现"回退到 `<shortHash>`（工作区改动保留）"；两段式确认（点击后按钮变红"确认回退？"，再点执行，点别处取消）——不用 `window.confirm`（GUI 内体验差）。
- 页签顺序不变（文件→变更→会话）；"变更"页签承载全部新功能（用户裁定，不新增页签）。

// lib/services/git-history.js —— git 提交历史 / 提交 / 安全回退（复用 git-diff.js 的 runGit）
import { stat } from "node:fs/promises";
import { runGit, gitBase } from "./git-diff.js";

const TARGET_RE = /^[0-9a-f]{4,40}$/;

async function ensureRepo(cwd) {
  await stat(cwd).catch(() => {
    throw { code: "dir-not-found", message: `目录不存在: ${cwd}` };
  });
}

export async function logCommits(cwd, { limit = 50 } = {}) {
  await ensureRepo(cwd);
  const out = await runGit(cwd, ["log", `-n ${limit}`, "--format=%H%x1f%h%x1f%s%x1f%an%x1f%aI"]);
  const commits = out
    .split("\n")
    .filter((line) => line !== "")
    .map((line) => {
      const [hash, shortHash, subject, author, date] = line.split("\x1f");
      return { hash, shortHash, subject, author, date };
    });
  return { commits };
}

export async function commitAll(cwd, message, files) {
  await ensureRepo(cwd);
  if (files && files.length > 0) {
    await runGit(cwd, ["add", "--", ...files]);
  } else {
    await runGit(cwd, ["add", "-A"]);
  }
  const staged = await runGit(cwd, ["diff", "--cached", "--quiet"]);
  // 注意：diff --cached --quiet 无可提交时 exit 0；有变更时 exit 1（runGit 视非零为失败）
  // 因此这里用 add 后的 diff --cached --name-only 判断
  const names = await runGit(cwd, ["diff", "--cached", "--name-only"]);
  if (names.trim() === "") throw { code: "nothing-to-commit", message: "没有可提交的变更" };
  const out = await runGit(cwd, ["commit", "-m", message]);
  const hashMatch = /^\[[^\]]+\] [0-9a-f]{40}/m.exec(out) ?? /[0-9a-f]{40}/.exec(out);
  const hash = hashMatch ? hashMatch[0].match(/[0-9a-f]{40}/)[0] : (await runGit(cwd, ["rev-parse", "HEAD"])).trim();
  return { hash };
}

export async function resetTo(cwd, target) {
  await ensureRepo(cwd);
  if (typeof target !== "string" || !TARGET_RE.test(target)) {
    throw { code: "invalid-target", message: `非法回退目标: ${target}` };
  }
  // 仅 mixed（安全回退：HEAD 回退、工作区改动保留）——用户裁定，不做 hard
  await runGit(cwd, ["reset", "--mixed", target]);
  return { ok: true };
}
```

> 说明：`git diff --cached --quiet` 的退出码语义（0=无差异）与 runGit 的失败判定冲突，故用 `git diff --cached --name-only` 判断可提交内容；`commitAll` 的 hash 从 commit 输出或 `rev-parse HEAD` 提取。

// src/lib/git-history-client.js —— 提交历史 client 纯逻辑（相对时间等）
export function relativeTime(iso, now = Date.now()) {
  const t = Date.parse(iso);
  if (Number.isNaN(t)) return "";
  const diff = Math.max(0, now - t);
  const min = Math.floor(diff / 60000);
  if (min < 1) return "刚刚";
  if (min < 60) return `${min} 分钟前`;
  const hr = Math.floor(min / 60);
  if (hr < 24) return `${hr} 小时前`;
  const day = Math.floor(hr / 24);
  if (day < 30) return `${day} 天前`;
  const mon = Math.floor(day / 30);
  if (mon < 12) return `${mon} 个月前`;
  return `${Math.floor(mon / 12)} 年前`;
}
```
